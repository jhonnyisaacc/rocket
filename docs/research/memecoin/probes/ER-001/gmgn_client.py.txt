from __future__ import annotations

import json
import logging
import os
import random
import shutil
import subprocess
import time
import uuid

from curl_cffi import requests as cf

logger = logging.getLogger(__name__)

BASE = "https://gmgn.ai"
IMPERSONATE = "chrome124"
GMGN_API_KEY = os.getenv("GMGN_API_KEY", "")
HAS_CLI = bool(GMGN_API_KEY) and shutil.which("gmgn-cli") is not None


def _sleep():
    time.sleep(random.uniform(1.0, 2.0))


def _common_params() -> dict:
    return {
        "device_id": str(uuid.uuid4()),
        "fp_did": "unknown",
        "client_id": f"gmgn_web_{int(time.time())}-646-8faad65",
        "from_app": "gmgn",
        "app_lang": "en-US",
        "os": "web",
        "worker": "0",
    }


def _get(url: str, params: dict, referer: str) -> dict | None:
    try:
        r = cf.get(url, params=params, headers={"Referer": referer},
                   impersonate=IMPERSONATE, timeout=15)
        _sleep()
        if r.status_code == 429:
            logger.warning("GMGN 429 rate limited, backing off 30s")
            time.sleep(30)
            return None
        if r.status_code == 200:
            d = r.json()
            if d.get("code") == 0:
                return d.get("data") or {}
        return None
    except Exception as exc:
        logger.debug("GMGN GET %s: %s", url[-50:], exc)
        return None


def _cli(cmd: str, timeout: int = 30) -> dict:
    if not HAS_CLI:
        return {}
    env = {**os.environ, "GMGN_API_KEY": GMGN_API_KEY}
    try:
        r = subprocess.run(f"gmgn-cli {cmd} --raw", capture_output=True,
                           env=env, shell=True, timeout=timeout)
        d = json.loads((r.stdout or r.stderr or b"").decode("utf-8", errors="replace"))
        return d if isinstance(d, dict) else {}
    except Exception as exc:
        logger.debug("gmgn-cli %s: %s", cmd.split()[0], exc)
        return {}


# -- discovery -----------------------------------------------------------

def discover_graduated(max_age_sec: int = 1800) -> list[dict]:
    d = _cli("market trenches --chain sol --type completed --limit 80") if HAS_CLI else {}
    completed = d.get("completed") or []
    now = int(time.time())
    out: dict[str, dict] = {}
    for t in completed:
        addr, comp_ts = t.get("address"), t.get("complete_timestamp") or 0
        if addr and comp_ts and now - comp_ts <= max_age_sec and addr not in out:
            out[addr] = {"address": addr, "symbol": t.get("symbol", ""), "source": "grad"}
    return list(out.values())


_TRENDING_COMBOS = [
    ("swaps", "1m"), ("swaps", "5m"), ("swaps", "1h"), ("swaps", "6h"),
    ("volume", "1h"), ("change1h", "1h"),
]


def discover_trending() -> list[dict]:
    seen: dict[str, dict] = {}
    for orderby, interval in _TRENDING_COMBOS:
        rank: list = []
        if HAS_CLI:
            d = _cli(f"market trending --chain sol --interval {interval} "
                     f"--order-by {orderby} --direction desc --limit 100")
            rank = (d.get("data") or {}).get("rank") or d.get("rank") or []
        else:
            d = _get(f"{BASE}/defi/quotation/v1/rank/sol/swaps/{interval}",
                     {**_common_params(), "orderby": orderby, "direction": "desc",
                      "limit": "100"},
                     f"{BASE}/trend?chain=sol")
            rank = (d or {}).get("rank") or []
        for t in rank:
            addr = t.get("address")
            if addr and addr not in seen:
                seen[addr] = {"address": addr, "symbol": t.get("symbol", ""), "source": "hot"}
    return list(seen.values())


def discover_tokens(grad_ratio: float = 0.7) -> list[dict]:
    grad = discover_graduated()
    grad_addrs = {t["address"] for t in grad}
    hot = [t for t in discover_trending() if t["address"] not in grad_addrs]
    if grad and 0.0 < grad_ratio < 1.0:
        hot = hot[: int(len(grad) * (1 - grad_ratio) / grad_ratio)]
    return grad + hot


# -- per-token features ----------------------------------------------------

def token_stat(addr: str) -> dict:
    d = _get(f"{BASE}/api/v1/token_stat/sol/{addr}", _common_params(),
             f"{BASE}/sol/token/{addr}")
    if d is None and HAS_CLI:
        info = _cli(f"token info --chain sol --address {addr}")
        stat = info.get("stat") or {}
        hc = int(float(info.get("holder_count") or stat.get("holder_count") or 0))
        if not (stat or hc):
            return {}
        return {
            "holder_count": hc,
            "top10_holder_pct": float(stat.get("top_10_holder_rate") or 0),
            "dev_remaining_pct": float(stat.get("creator_hold_rate") or 0),
            "rat_position_pct": float(stat.get("top_rat_trader_percentage") or 0),
            "new_wallet_count": int(float(stat.get("fresh_wallet_rate") or 0) * max(hc, 1)),
            "liquidity_usd": float(info.get("liquidity") or 0),
        }
    if not d:
        return {}
    hc = int(d.get("holder_count") or 0)
    return {
        "holder_count": hc,
        "top10_holder_pct": float(d.get("top_10_holder_rate") or 0),
        "dev_remaining_pct": float(d.get("creator_hold_rate") or 0),
        "rat_position_pct": float(d.get("top_rat_trader_percentage") or 0),
        "new_wallet_count": int(float(d.get("fresh_wallet_rate") or 0) * max(hc, 1)),
    }


def wallet_tags(addr: str) -> dict:
    d = _get(f"{BASE}/api/v1/token_wallet_tags_stat/sol/{addr}", _common_params(),
             f"{BASE}/sol/token/{addr}")
    if d is None and HAS_CLI:
        info = _cli(f"token info --chain sol --address {addr}")
        d = info.get("wallet_tags_stat") or {}
    if not d:
        return {}
    return {
        "smart_money_count": int(d.get("smart_wallets") or 0),
        "sniper_count": int(d.get("sniper_wallets") or 0),
    }


def security(addr: str) -> dict:
    d = _get(f"{BASE}/api/v1/mutil_window_token_security_launchpad/sol/{addr}",
             _common_params(), f"{BASE}/sol/token/{addr}")
    sec = (d or {}).get("security") or {}
    if not sec and HAS_CLI:
        sec = _cli(f"token security --chain sol --address {addr}")
    if not sec:
        return {}
    return {
        "mint_disabled": bool(sec.get("renounced_mint")),
        "freeze_disabled": bool(sec.get("renounced_freeze_account")),
        "lp_burned_pct": float(sec.get("burn_ratio") or 0),
        "is_honeypot": float(sec.get("is_honeypot") or sec.get("honeypot") or 0),
    }


def links(addr: str) -> dict:
    d = _get(f"{BASE}/api/v1/mutil_window_token_link_rug_vote/sol/{addr}",
             _common_params(), f"{BASE}/sol/token/{addr}")
    if not d:
        return {}
    link, rug = d.get("link") or {}, d.get("rug") or {}
    return {
        "has_twitter": bool((link.get("twitter_username") or "").strip()),
        "has_website": bool((link.get("website") or "").strip()),
        "has_telegram": bool((link.get("telegram") or "").strip()),
        "rug_vote_ratio": float(rug.get("rug_ratio") or 0),
    }


# -- trades (graph construction / behavior features) -----------------------

SMART_TAGS = {"smart_degen", "smart_money", "pump_smart", "renowned", "bluechip_owner"}
BUNDLER_TOKEN_TAGS = {"bundler", "sniper"}


def token_trades(addr: str, limit: int = 200) -> list[dict]:
    params = {**_common_params(), "limit": str(limit), "maker": ""}
    d = _get(f"{BASE}/vas/api/v1/token_trades/sol/{addr}", params,
             f"{BASE}/sol/token/{addr}")
    history = (d or {}).get("history") or []
    out = []
    for tx in history:
        w = tx.get("maker", "")
        if not w or len(w) < 32:
            continue
        out.append({
            "wallet": w,
            "side": "buy" if str(tx.get("event", "buy")).lower() in ("buy", "b") else "sell",
            "amount_usd": float(tx.get("amount_usd") or 0),
            "ts": int(tx.get("timestamp") or 0),
            "tags": tx.get("maker_tags") or [],
            "token_tags": tx.get("maker_token_tags") or [],
            "realized_profit": float(tx.get("realized_profit") or 0),
            "balance": float(tx.get("balance") or 0),
        })
    if out:
        return out
    # CLI fallback aggregates per-wallet totals instead of a trade log, so we
    # synthesize one buy + one sell row per wallet (timestamps are approximate)
    if HAS_CLI:
        d = _cli(f"token traders --chain sol --address {addr}")
        for tr in (d.get("list") or [])[:limit]:
            w = tr.get("address") or ""
            if not w or len(w) < 32 or (tr.get("addr_type") or 0) != 0:
                continue
            base = {"wallet": w, "tags": list(tr.get("tags") or []),
                    "token_tags": [k for k in ("bundler", "sniper", "top_holder", "dev")
                                   if k in set(tr.get("maker_token_tags") or [])],
                    "realized_profit": float(tr.get("realized_profit") or 0),
                    "balance": float(tr.get("amount_cur") or 0)}
            t0 = int(tr.get("start_holding_at") or 0)
            t1 = int(tr.get("end_holding_at") or t0)
            buy_usd = float(tr.get("buy_volume_cur") or 0)
            sell_usd = float(tr.get("sell_volume_cur") or 0)
            if buy_usd > 0:
                out.append({**base, "side": "buy", "amount_usd": buy_usd, "ts": t0})
            if sell_usd > 0:
                out.append({**base, "side": "sell", "amount_usd": sell_usd, "ts": t1})
    return out


# -- candles (labeling) -----------------------------------------------------

def mcap_candles(addr: str, before_ts: int, resolution: str = "5m",
                 limit: int = 1000) -> list[list]:
    params = {**_common_params(), "resolution": resolution, "limit": str(limit),
              "before_timestamp": str(before_ts), "pool_type": "tpool"}
    d = _get(f"{BASE}/api/v1/token_mcap_candles/sol/{addr}", params,
             f"{BASE}/sol/token/{addr}")
    rows = []
    for c in (d or {}).get("list") or []:
        try:
            ts = int(c.get("time") or 0) // 1000
            o, h, l, cl = (float(c.get(k) or 0) for k in ("open", "high", "low", "close"))
            if ts and o > 0 and cl > 0:
                rows.append([ts, o, h, l, cl, float(c.get("volume") or 0)])
        except (TypeError, ValueError):
            continue
    rows.sort(key=lambda r: r[0])
    return rows
